## Go Solution

```go
func postorder(root *Node) []int {
    var res []int
    var dfs func(node *Node)
    dfs = func(node *Node) {
        if node == nil {
            return
        }

        for _,child := range(node.Children) {
            dfs(child)
        }
        res = append(res, node.Val)
    }
    dfs(root)
    return res
}
```
