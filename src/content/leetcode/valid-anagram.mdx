## Go Solution

```go
func isAnagram(s string, t string) bool {
  if len(s) != len(t) {
    return false
  }

  m := make(map[rune]int)
  for _, letter := range s {
    m[letter]++
  }

  for _, letter := range t {
    m[letter]--
    if m[letter] < 0 {
      return false
    }
  }

  return true
}
```
