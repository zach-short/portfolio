```go
func hasCycle(head *ListNode) bool {
    m := make(map[*ListNode]bool)

    node := head
    for node != nil {
        if m[node] {
            return true
        }
        m[node] = true
        node = node.Next
    }

    return false
}
```

## Go Solution

```go
func hasCycle(head *ListNode) bool {
    slow,fast := head,head
    for fast != nil && fast.Next != nil {
        slow = slow.Next
        fast = fast.Next.Next
        if slow == fast {
            return true
        }
    }

    return false
}
```
