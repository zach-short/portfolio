## Go Solution

```go
func removeElement(nums []int, val int) int {
  c := 0
  for i := 0; i <= len(nums)-1-c; i++ {
    for nums[i] == val && i <= lch(nums,c) {
      if i != lch(nums, c) {
        nums[i], nums[lch(nums, c)] = nums[lch(nums, c)], nums[i]
      }
      c++
    }
  }
  return lch(nums, c) + 1
}

func lch(n []int, c int) int {
  return len(n) - 1 - c
}
```
