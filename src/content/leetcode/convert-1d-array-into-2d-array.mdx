## Go Solution

```go
func construct2DArray(original []int, m int, n int) [][]int {
    if m * n != len(original) {
        return [][]int{}
    }

    res := [][]int{}
    for i := 0; i < m; i++ {
        res = append(res, original[n*i:i*n+n])
    }
    return res
}


```
